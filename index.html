// supabase/functions/extract-invoice/index.ts // Reads an uploaded invoice (image or PDF) with Claude and pulls out the // vendor's bank account number / IFSC plus a few other fields, so the // WeRoCon app can auto-fill the Payment Reimbursement Form. // // Deploy: supabase functions deploy extract-invoice // Secret: supabase secrets set ANTHROPIC_API_KEY=sk-ant-... const ANTHROPIC_API_KEY = Deno.env.get("ANTHROPIC_API_KEY"); Deno.serve(async (req) => { const cors = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", }; if (req.method === "OPTIONS") return new Response("ok", { headers: cors }); try { if (!ANTHROPIC_API_KEY) { return new Response(JSON.stringify({ error: "ANTHROPIC_API_KEY not set in Supabase secrets" }), { status: 500, headers: cors }); } const { fileBase64, mediaType } = await req.json(); if (!fileBase64 || !mediaType) { return new Response(JSON.stringify({ error: "fileBase64 and mediaType are required" }), { status: 400, headers: cors }); } const isPdf = mediaType === "application/pdf"; const contentBlock = isPdf ? { type: "document", source: { type: "base64", media_type: "application/pdf", data: fileBase64 } } : { type: "image", source: { type: "base64", media_type: mediaType, data: fileBase64 } }; const prompt = `Look at this invoice/bill. Extract the following fields and reply with ONLY a raw JSON object — no markdown fences, no preamble, no explanation: { "invoiceNumber": "", "invoiceDate": "DD/MM/YYYY, or empty string if not found", "amount": 0, "vendorName": "", "vendorBankAccount": "", "vendorIfsc": "" } Important: vendorBankAccount and vendorIfsc must belong to the VENDOR / SELLER (the company that should receive payment) — not the buyer/institute. If a field isn't present on the invoice, leave it as an empty string (or 0 for amount). Do not guess or invent values.`; const res = await fetch("https://api.anthropic.com/v1/messages", { method: "POST", headers: { "content-type": "application/json", "x-api-key": ANTHROPIC_API_KEY, "anthropic-version": "2023-06-01", }, body: JSON.stringify({ model: "claude-sonnet-4-6", max_tokens: 500, messages: [{ role: "user", content: [contentBlock, { type: "text", text: prompt }] }], }), }); const data = await res.json(); if (!res.ok) { console.error("Anthropic API error", res.status, data); return new Response(JSON.stringify({ error: data?.error?.message || `Anthropic API returned HTTP ${res.status}` }), { status: 500, headers: cors }); } const text = (data.content || []).map((b) => b.text || "").join("").trim(); const cleaned = text.replace(/```json|```/g, "").trim(); let parsed; try { parsed = JSON.parse(cleaned); } catch { console.error("Could not parse extraction JSON", text); return new Response(JSON.stringify({ error: "Could not parse extraction result", raw: text }), { status: 500, headers: cors }); } return new Response(JSON.stringify(parsed), { headers: { ...cors, "content-type": "application/json" } }); } catch (e) { console.error("extract-invoice failed", e); return new Response(JSON.stringify({ error: e.message }), { status: 500, headers: cors }); } });
